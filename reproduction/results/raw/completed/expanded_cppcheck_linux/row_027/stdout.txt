from z3 import *

# Declare variable `window` as an integer (C-style scalar)
window = Int('window')

# Solver instance
s = Solver()

# Constraint: !(window == 0)
# In C, `!(window == 0)` is equivalent to `window != 0`.
# This constraint is included because it directly restricts the value of `window`.
s.add(window != 0)

# Check satisfiability and print result
print(s.check())
print(s.model())
