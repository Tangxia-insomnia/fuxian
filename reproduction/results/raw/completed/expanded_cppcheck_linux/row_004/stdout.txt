from z3 import *

# Declare variable `res` as an integer (C-style variable)
res = Int('res')

# Solver instance
s = Solver()

# Constraint: !(res == 0)
# In C, this means res is non-zero (i.e., res != 0).
# Converted to Z3 as: res != 0

# Check satisfiability and print result
print(s.check())
if s.check() == sat:
    print(s.model())
