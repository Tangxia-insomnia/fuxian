from z3 import *

# Declare variable mpic as an integer (C-style scalar)
mpic = Int('mpic')

# Constraint: !(mpic == 0)
# In C, this means mpic is non-zero (i.e., mpic != 0).
# Converted to Z3 as: mpic != 0
s = Solver()
s.add(mpic != 0)

# Check satisfiability and print a model if satisfiable
if s.check() == sat:
    print(s.model())
else:
    print("unsat")
