from z3 import *

# Declare variables appearing in the constraint
# elem_type: result of get_bits(gb, 3) - an integer bitfield value
elem_type = Int('elem_type')
# TYPE_END and TYPE_DSE are C enum constants; model as symbolic integers
TYPE_END = Int('TYPE_END')
TYPE_DSE = Int('TYPE_DSE')
# che: result of get_che(...) - modeled as an integer (0 means NULL/failure)
che = Int('che')

s = Solver()

# Original C constraint:
# !((elem_type = get_bits(gb, 3)) != TYPE_END) || !(elem_type < TYPE_DSE) || !(!(che=get_che(ac, elem_type, elem_id)))
#
# Simplify step by step:
#   A = (elem_type != TYPE_END)          [from (elem_type = get_bits(...)) != TYPE_END]
#   B = (elem_type < TYPE_DSE)
#   C = (che != 0)                        [from !(che = get_che(...)) i.e. che is non-NULL]
#   Constraint = !A || !B || !(!C)
#              = !A || !B || C
#
# So the constraint is: (elem_type == TYPE_END) || (elem_type >= TYPE_DSE) || (che != 0)

# Included because it directly constrains elem_type and che, the variables of interest.
s.add(Or(elem_type == TYPE_END,
         elem_type >= TYPE_DSE,
         che != 0))

# Print the solver state for inspection
print(s)
print(s.check())
