from z3 import *

# Declare variables appearing in the constraints
config = Bool('config')
i = Int('i')

# The input constraint is: !(i = 0;; i++) || !(!config)
# The C for-loop header "i = 0;; i++" is not a boolean expression; it is
# malformed when used with logical negation. The only meaningful, well-formed
# sub-expression that affects the listed variable `config` is `!(!config)`.
# We therefore model the constraint as: !(!config)  (i.e., config must be true)
# The `i` variable is declared for completeness but does not appear in any
# well-formed boolean sub-expression, so no constraint is placed on it.

s = Solver()

# Constraint: !(!config)  ->  config == True
# Included because it is the only well-formed boolean condition referencing
# the relevant variable `config`.
s.add(Not(Not(config)))

# Check satisfiability and print result
print(s.check())
print(s.model())
