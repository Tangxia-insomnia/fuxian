# sub->rects is modeled as an array of pointers (Ints); a null pointer is represented by 0.

sub_rects = Array('sub_rects', IntSort(), IntSort())

#   !(sub->num_rects > 0) || !(i = 0; i < sub->num_rects; i++) || !(!sub->rects[i])
# The middle clause is a C for-loop header, which is not a boolean expression.
# Its intent is: for all i in [0, sub->num_rects), sub->rects[i] must be non-null.
# Combined with the first clause (num_rects <= 0 short-circuits), the constraint
# reduces to: if num_rects > 0, then every rects[i] for 0 <= i < num_rects is non-null.
# We encode this as: num_rects <= 0 OR (for all i, 0 <= i < num_rects => rects[i] != 0).
i = Int('i')
constraint1 = Or(
    sub_num_rects <= 0,
    ForAll(i, Implies(And(i >= 0, i < sub_num_rects), Select(sub_rects, i) != 0))
)
