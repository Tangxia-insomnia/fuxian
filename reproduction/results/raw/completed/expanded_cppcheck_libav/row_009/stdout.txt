from z3 import *

# Declare variables used in the constraint
# pkt is a pointer; model as an integer (0 = NULL, non-zero = valid pointer)
pkt = Int('pkt')

# st->codecpar->codec_id is compared against AV_CODEC_ID_PROBE.
# We model the boolean "codec_id == AV_CODEC_ID_PROBE" as a symbolic Bool.
codec_id_is_probe = Bool('codec_id_is_probe')

# av_reallocp returns an int; the constraint checks (err < 0).
# Model err as an Int and the call result as a symbolic Int.
err = Int('err')

# The constraint:
#   !(st->codecpar->codec_id == AV_CODEC_ID_PROBE) || !(pkt) || !((err = av_reallocp(...)) < 0)
#
# In C, `!(pkt)` is true when pkt is NULL (i.e., pkt == 0).
# `!(err < 0)` is true when err >= 0.
# The whole expression is a disjunction; we assert it as a constraint.
#
# Included because it is the only constraint given and it references `pkt`.

constraint = Or(
    Not(codec_id_is_probe),   # codec_id != AV_CODEC_ID_PROBE
    pkt == 0,                 # pkt is NULL
    err >= 0                  # av_reallocp did not fail (err >= 0)
)

s = Solver()
s.add(constraint)

# Check satisfiability and print a model if satisfiable
print(s.check())
if s.check() == sat:
    print(s.model())
