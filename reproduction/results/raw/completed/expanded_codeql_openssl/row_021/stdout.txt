from z3 import *

# Declare variables
# cmd is a C string (pointer to char). We model it as a symbolic string.
cmd = String('cmd')

# The constraint involves strchr(cmd, ':') which returns a pointer to the first
# occurrence of ':' in cmd, or NULL if not found.
# In C, `arg = strchr(cmd, ':')` and the check `!(arg == NULL)` means arg != NULL,
# i.e., ':' IS present in cmd.
# We model this with Z3's string containment: Contains(cmd, ":").

# The constraint also involves `(int)(arg - cmd) > 254`, i.e., the offset of ':'
# from the start of cmd is greater than 254. This means the first ':' appears
# at index > 254 (i.e., at least 255 characters into the string).
# We model this with IndexOf(cmd, ":") > 254.

# The loop constraint `!(loop = 0; loop < num; loop++)` is a negation of a
# for-loop condition. In C, `!(loop = 0; loop < num; loop++)` is not a valid
# boolean expression (it's a syntax error / always-false construct in a boolean
# context). Interpreting it as the negation of the loop guard `loop < num`
# (with loop initialized to 0), the negation is `!(0 < num)` i.e. `num <= 0`.
# Since `num` is not in the relevant variables list, and the guideline says to
# convert only constraints affecting the specified conditions/variables (cmd),
# we focus on the parts involving `cmd`.

# The full C constraint is a disjunction:
#   !(loop guard) || !(arg != NULL) || !((int)(arg - cmd) > 254)
# which simplifies to:
#   (loop guard false) OR (arg == NULL) OR ((int)(arg - cmd) <= 254)

# Since we only care about constraints affecting `cmd`, we keep the two
# cmd-related disjuncts. The constraint is satisfiable if EITHER:
#   - ':' is NOT in cmd (arg == NULL), OR
#   - ':' IS in cmd AND its first index is <= 254.

s = Solver()

# Constraint affecting cmd:
# The disjunction of the two cmd-related conditions must hold.
# (arg == NULL)  OR  (IndexOf(cmd, ":") <= 254)
# Note: if ':' is not in cmd, IndexOf returns -1, which is <= 254, so the
# second disjunct alone would be too weak. We must keep the structure:
#   Not(Contains(cmd, ":"))  OR  (Contains(cmd, ":") AND IndexOf(cmd, ":") <= 254)
# which simplifies to: IndexOf(cmd, ":") <= 254 (since -1 <= 254 when absent).
# However, to faithfully mirror the C semantics (arg==NULL short-circuits),
# we write it explicitly.

arg_is_null = Not(Contains(cmd, ":"))
arg_offset_ok = And(Contains(cmd, ":"), IndexOf(cmd, ":") <= 254)

s.add(Or(arg_is_null, arg_offset_ok))

# Print the solver state
print(s)
print(s.check())
if s.check() == sat:
    print(s.model())
