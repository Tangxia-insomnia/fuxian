# `in` is a pointer; model it as an integer address (0 == NULL).
in_ptr = Int('in')
# `passed_salt` is a pointer; model as integer address (0 == NULL).

in_noverify = Bool('in_noverify')

# `do_passwd(...)` is an opaque C function returning a boolean-ish int.
# Model its return value as an uninterpreted boolean.
do_passwd_ret = Bool('do_passwd_ret')

# `EVP_read_pw_string(...)` returns an int; the constraint checks `!= 0`.
# Model its return value as an uninterpreted integer.
evp_read_pw_string_ret = Int('evp_read_pw_string_ret')

s = Solver()

# Constraint 1:
#   !(in == NULL) || !(!do_passwd(...))
# In C: if `in` is non-NULL, then `do_passwd(...)` must be truthy.

s.add(Implies(in_ptr != 0, do_passwd_ret))

# Rewritten: (in != 0) => (evp_read_pw_string_ret == 0)
s.add(Implies(in_ptr != 0, evp_read_pw_string_ret == 0))

    passed_salt = Int('passed_salt')
