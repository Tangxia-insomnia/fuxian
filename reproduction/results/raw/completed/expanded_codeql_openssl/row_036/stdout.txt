from z3 import *

# Declarations for the variables appearing in the constraints.
# We model pointers as integer addresses (0 == NULL) and booleans as Bool.
in_ptr = Int('in')                 # pointer 'in' (0 means NULL)
passed_salt = Bool('passed_salt')  # boolean flag
in_noverify = Bool('in_noverify')  # boolean flag

# The function calls do_passwd(...) and EVP_read_pw_string(...) are opaque
# to the solver, so we introduce uninterpreted boolean predicates to
# represent their return values / success status.
do_passwd_ok = Bool('do_passwd_ok')          # true if do_passwd(...) succeeds
evp_read_pw_ok = Bool('evp_read_pw_ok')      # true if EVP_read_pw_string(...) == 0

s = Solver()

# Constraint 1:
#   !(in == NULL) || !(!do_passwd(...))
# In C, !do_passwd(...) is true when do_passwd returns 0 (failure).
# So !(!do_passwd(...)) means do_passwd(...) returned non-zero (success).
# Rewritten: (in != NULL) || do_passwd_ok
# This constraint is included because it directly references 'in' and
# gates the password-processing path on the do_passwd call succeeding.
s.add(Or(in_ptr != 0, do_passwd_ok))

# Constraint 2:
#   !(in == NULL) || !(EVP_read_pw_string(...) != 0)
# EVP_read_pw_string(...) != 0 means failure; negating gives success.
# Rewritten: (in != NULL) || evp_read_pw_ok
# Included because it references 'in' and gates the password-reading path
# on EVP_read_pw_string succeeding.
s.add(Or(in_ptr != 0, evp_read_pw_ok))

# Note: 'passwd' is the buffer written by these calls; it does not appear
# as a free variable in the given constraints, so no separate declaration
# is required for the solver.

print(s)

def check_constraints():
    from z3 import Int, Bool, Solver, Or, sat

    # Base script variables and constraints
    passwd = Int('passwd')

    # Variables from constraint set 2
    in_ptr = Int('in')
    passed_salt = Bool('passed_salt')
    in_noverify = Bool('in_noverify')
    do_passwd_ok = Bool('do_passwd_ok')
    evp_read_pw_ok = Bool('evp_read_pw_ok')

    s = Solver()

    # Base constraint
    s.add(passwd == 0)

    # Constraint set 2 constraints
    s.add(Or(in_ptr != 0, do_passwd_ok))
    s.add(Or(in_ptr != 0, evp_read_pw_ok))

    return s.check() == sat
