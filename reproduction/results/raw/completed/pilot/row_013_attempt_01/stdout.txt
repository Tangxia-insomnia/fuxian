s.add(Not(tcp_clone_payload(sk, nskb, probe_size)))

# ---------------------------------------------------------------------------
s.add(nskb != 0)

#   || !(!(!tcp_packets_in_flight(tp)))

# Kept because it participates in the same guard chain as the skb check.
# ---------------------------------------------------------------------------
s.add(Or(
    Not(tcp_packets_in_flight(tp) + 2 > tcp_snd_cwnd(tp)),
    tcp_packets_in_flight(tp) == 0

    tcp_packets_in_flight(tp) != 0
