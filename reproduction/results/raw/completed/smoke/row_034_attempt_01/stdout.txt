fc_flags = Int('fc_flags')
dev = Int('dev')

loopback_dev = Int('loopback_dev')

s = Solver()

# C-style translation:
#   !(!idev)  ==>  idev != 0
#   !(dev != net->loopback_dev)  ==>  dev == loopback_dev
