tun_info = BitVec('tun_info', 64)

# Constants from the C headers.
IP_TUNNEL_INFO_TX = BitVecVal(0x1, 32)   # bit flag in tun_info->mode
AF_INET           = BitVecVal(2, 16)     # Linux AF_INET value
ETH_P_IP          = BitVecVal(0x0800, 16)
ETH_P_IPV6        = BitVecVal(0x86DD, 16)

#
#   !(!(tun_info
#        && (tun_info->mode & IP_TUNNEL_INFO_TX)
#        && ip_tunnel_info_af(tun_info) == AF_INET
#        && tun_info->key.u.ipv4.dst)
#     && !(payload_protocol == htons(ETH_P_IP))
#     && !(payload_protocol == htons(ETH_P_IPV6)))

# The outer '!' plus the inner '!' on the first conjunct cancel out, so the
# constraint reduces to the conjunction of:
#   (a) tun_info != NULL
#   (b) (tun_info->mode & IP_TUNNEL_INFO_TX) != 0

#   (e) payload_protocol != htons(ETH_P_IP)
#   (f) payload_protocol != htons(ETH_P_IPV6)

def htons(x):

    return ((x & 0xFF) << 8) | ((x >> 8) & 0xFF)
