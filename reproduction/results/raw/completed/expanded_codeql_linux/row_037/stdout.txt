# The constraints reference `nh` (a nexthop pointer) and its field `fib_nh_flags`.
# We model `nh` as a pointer (Int) where 0 means NULL, and `fib_nh_flags` as a bitvector

RTNH_F_DEAD = BitVecVal(1, 32)

# uninterpreted booleans tied to `nh` being non-NULL (since they dereference nh).
nexthop_is_blackhole = Function('nexthop_is_blackhole', IntSort(), BoolSort())
unlikely_nh = Function('unlikely_nh', IntSort(), BoolSort())

#   !(unlikely(f6i->nh)) || !(nexthop_is_blackhole(f6i->nh))

c1 = Or(Not(unlikely_nh(nh)), Not(nexthop_is_blackhole(nh)))

#   !(!oif && ipv6_addr_any(saddr)) || !(!(nh->fib_nh_flags & RTNH_F_DEAD))
# Included because it references nh->fib_nh_flags (a field of `nh`).

# Right disjunct: !(!(nh->fib_nh_flags & RTNH_F_DEAD))

c2 = Or(left2, right2)

def check_constraints():

    # Base script constraints
    nh = Int('nh')

    # Constraint set 2 variable definitions
    fib_nh_flags = BitVec('fib_nh_flags', 32)
    RTNH_F_DEAD = BitVecVal(1, 32)
    nexthop_is_blackhole = Function('nexthop_is_blackhole', IntSort(), BoolSort())
    unlikely_nh = Function('unlikely_nh', IntSort(), BoolSort())

    s = Solver()

    # Base script constraint
    s.add(nh == 0)

    # Constraint 1
    c1 = Or(Not(unlikely_nh(nh)), Not(nexthop_is_blackhole(nh)))
    s.add(c1)

    # Constraint 2

    c2 = Or(left2, right2)
    s.add(c2)

    # Constraint 3
    left3 = Bool('left3')

    s.add(c3)

    return s.check() == sat
