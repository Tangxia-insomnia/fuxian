#
# Analysis:
#   - WARN_ON_ONCE(cond) is a Linux kernel macro that evaluates its argument
#     and returns the (boolean) result of that argument. It has no effect on
#     program semantics beyond emitting a warning; the expression value is
#     exactly the value of `cond`.
#   - Therefore WARN_ON_ONCE(!sband) evaluates to (!sband).
#   - The outer `!` negates it: !(WARN_ON_ONCE(!sband)) == !(!sband) == sband.
#   - In C, `sband` is a pointer; `!sband` is true iff sband == NULL.
#     So the constraint reduces to: sband != NULL (i.e., sband != 0).
#
# This constraint is included because it is the only condition that
# constrains the variable `sband` (forcing it to be non-NULL).

from z3 import *
