from z3 import *

# Variable declaration
# 'sta' is treated as a boolean condition (e.g., a pointer or integer used in a truth test).
# In C, WARN_ON(!sta) triggers a warning when !sta is true, i.e., when sta is falsy (0/NULL).
sta = Bool('sta')

# Constraint: !(WARN_ON(!sta))
# The outer '!' negates the WARN_ON result. WARN_ON(x) evaluates to true when x is true.
# So WARN_ON(!sta) is true iff !sta is true, i.e., iff sta is false.
# Negating it means we require WARN_ON(!sta) to be false, i.e., !sta must be false,
# which simplifies to: sta must be true.
# This constraint is included because it directly restricts the value of 'sta'.
s = Solver()
s.add(Not(Not(sta)))  # equivalent to sta == True

# Check satisfiability and print a model
print(s.check())
if s.check() == sat:
    print(s.model())
