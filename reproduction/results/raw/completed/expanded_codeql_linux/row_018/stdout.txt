def check_constraints():
    from z3 import Int, Solver, And, sat

    s = Solver()

    # Declare all variables as Int to avoid NameError / type mismatches
    fib6_ref = Int('fib6_ref')
    fn = Int('fn')
    fn_flags = Int('fn_flags')
    RTN_RTINFO = Int('RTN_RTINFO')
    leaf = Int('leaf')
    rt = Int('rt')
    new_leaf = Int('new_leaf')

    # Line 12: if (refcount_read(&rt->fib6_ref) != 1) - entering the branch
    s.add(fib6_ref != 1)

    # Line 19: while (fn) - loop condition true, fn is non-null
    s.add(fn != 0)

    # RTN_RTINFO flag bit value
    s.add(RTN_RTINFO == 1)

    # Line 23: !(fn->fn_flags & RTN_RTINFO) means bitwise AND result is 0
    # Use Z3 bitwise AND on Int via BitVec conversion is not allowed here;
    # instead model the masked-bit condition directly as an Int constraint.
    # Since RTN_RTINFO == 1, (fn_flags & 1) == 0 is equivalent to fn_flags being even.
    s.add(fn_flags % 2 == 0)

    # leaf == rt
    s.add(leaf == rt)

    # Assume pointer new_leaf is NULL
    s.add(new_leaf == 0)

    return s.check() == sat
