# Original C constraint: !((sc = SSL_CONNECTION_FROM_CONST_SSL(ssl)) == NULL)
#
# Analysis:
#   - SSL_CONNECTION_FROM_CONST_SSL(ssl) is a macro/function that returns a pointer
#     (SSL_CONNECTION*) derived from the SSL object. In Z3 we model its result as
#     an uninterpreted function returning an uninterpreted sort (pointer-like).
#   - The assignment `sc = ...` binds `sc` to that returned pointer.
#   - The comparison `(... == NULL)` checks whether the returned pointer is NULL.
#   - The outer `!` negates it, so the constraint asserts the pointer is NOT NULL.
#   - Therefore the effective constraint on `sc` is: sc != NULL.
#
# We model NULL as a distinguished constant of the pointer sort, and
# SSL_CONNECTION_FROM_CONST_SSL as an uninterpreted function from SSL to pointer.

from z3 import *

# Sort representing SSL objects (opaque)
SSL = DeclareSort('SSL')

# Sort representing SSL_CONNECTION pointers (opaque)
SSL_CONNECTION = DeclareSort('SSL_CONNECTION')

# NULL pointer constant
NULL = Const('NULL', SSL_CONNECTION)

# Uninterpreted function modeling the macro SSL_CONNECTION_FROM_CONST_SSL(ssl)
SSL_CONNECTION_FROM_CONST_SSL = Function(
    'SSL_CONNECTION_FROM_CONST_SSL', SSL, SSL_CONNECTION
)

# The SSL object referenced in the original expression
ssl = Const('ssl', SSL)

# The variable `sc` (result of the assignment)
sc = Const('sc', SSL_CONNECTION)

s = Solver()

# Constraint 1: sc equals the pointer returned by SSL_CONNECTION_FROM_CONST_SSL(ssl).
# This captures the assignment `sc = SSL_CONNECTION_FROM_CONST_SSL(ssl)`.
s.add(sc == SSL_CONNECTION_FROM_CONST_SSL(ssl))

# Constraint 2: the negated NULL check, i.e. !(sc == NULL)  <=>  sc != NULL.
# This is the actual assertion enforced by the original C expression.
s.add(sc != NULL)

# Check satisfiability and print result
print(s.check())
